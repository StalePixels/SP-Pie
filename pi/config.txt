# SP-Pie video service, Raspberry Pi Zero 2 W
#
# Copy to the FAT partition of the SD card as config.txt, together with
# kernel8.img and the firmware files listed in README.md.

# Start the ARM cores in 64-bit mode (AArch64) and load the Circle kernel
# at the address it is linked for.
arm_64bit=1
kernel=kernel8.img
kernel_address=0x80000
initial_turbo=0

# Run the ARM cores at their full clock (1000 MHz on the Zero 2 W) all the
# time. Without it the firmware leaves them at 600 MHz.
force_turbo=1

# HDMI on even when no display is detected at start-up.
hdmi_force_hotplug=1

# 1920 x 1080 at the 128K-class Spectrum frame rate, 3546900 / 70908 =
# 50.0211 Hz: the CEA 1080p50 timings (CEA-861, mode 31) with the pixel
# clock raised from 148.5 MHz to 148.5 * 50.0211 / 50 MHz. Format of
# hdmi_timings: h_active h_sync_pol h_front h_sync h_back v_active v_sync_pol
# v_front v_sync v_back v_offset_a v_offset_b pixel_rep frame_rate interlaced
# pixel_freq aspect_ratio (3 = 16:9).
hdmi_group=2
hdmi_mode=87
hdmi_timings=1920 1 528 44 148 1080 1 4 5 36 0 0 0 50 0 148562661 3

# HDMI mode, not DVI, so the link carries the Spectrum's sound.
hdmi_drive=2
